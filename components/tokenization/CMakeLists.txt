cmake_minimum_required(VERSION 3.28)
project(CelleratorTokenization LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CELLERATOR_ENABLE_MODULES "Build the tokenizer module" OFF)
option(CELLERATOR_BUILD_TESTS "Build the native tokenizer checks" ON)
set(CELLERATOR_ENABLE_TOKENIZATION ON)
set(CELLERATOR_ENABLE_CUDA "AUTO" CACHE STRING "Enable the optional CUDA tokenizer implementation")
find_package(Cellerator CONFIG REQUIRED COMPONENTS annotation)
if(CELLERATOR_ENABLE_MODULES)
    include(../../cmake/CelleratorModuleInterfaces.cmake)
    cellerator_require_module_scanner()
endif()
include(targets.cmake)
