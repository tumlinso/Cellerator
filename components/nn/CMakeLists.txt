cmake_minimum_required(VERSION 3.28)
project(CelleratorNN LANGUAGES CXX)
include(GNUInstallDirs)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CELLERATOR_ENABLE_NN_CUDA "Build the CUDA NN adapters" OFF)
option(CELLERATOR_ENABLE_MODULES "Build the NN module" OFF)
set(CELLERATOR_ENABLE_NN ON)
find_package(Cellerator CONFIG REQUIRED)
find_package(CelleratorProduct2 CONFIG REQUIRED)
if(CELLERATOR_ENABLE_MODULES)
    include(../../cmake/CelleratorModuleInterfaces.cmake)
    cellerator_require_module_scanner()
    install(FILES ../../cmake/CelleratorModuleInterfaces.cmake
        DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Cellerator)
endif()
if(CELLERATOR_ENABLE_NN_CUDA)
    enable_language(CUDA)
    find_package(Cellerator CONFIG REQUIRED COMPONENTS framework_cuda factorized_interaction_cuda)
endif()
include(targets.cmake)
