cmake_minimum_required(VERSION 3.24)

if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 70 CACHE STRING "CUDA architectures")
endif()
project(cellerator_chromatin_geometry LANGUAGES C CXX CUDA)

find_package(CUDAToolkit REQUIRED)
find_package(HDF5 QUIET COMPONENTS C)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
include(CTest)

get_filename_component(_cgs_default_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CELLERATOR_ROOT "${_cgs_default_root}" CACHE PATH
    "Cellerator checkout or extracted immutable source closure")

# CGS-10 scaffold: CPU-only executable, with no CUDA calls or device context.
# Numerical/learning/runner targets are added as their owners deliver sources.
add_executable(cgs_contract_smoke contract_smoke.cc)
target_compile_features(cgs_contract_smoke PRIVATE cxx_std_20)
target_include_directories(cgs_contract_smoke PRIVATE
    "${CELLERATOR_ROOT}/include" "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(cgs_contract_smoke PRIVATE CUDA::cudart)
add_executable(cgs_study main.cc body_runner.cc campaign_runner.cu data/replay.cc)
target_compile_features(cgs_study PRIVATE cxx_std_20)

add_library(cgs_ste STATIC learning/ste.cu learning/ste_chain.cu learning/model_update.cu)
target_compile_features(cgs_ste PUBLIC cxx_std_20 cuda_std_20)
target_include_directories(cgs_ste PUBLIC
    "${CELLERATOR_ROOT}/include" "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(cgs_ste PUBLIC CUDA::cudart)
add_executable(cgs_ste_host_tests learning/ste_host_tests.cc)
target_link_libraries(cgs_ste_host_tests PRIVATE cgs_ste)
add_executable(cgs_ste_gpu_tests learning/ste_gpu_test_main.cc learning/ste_tests.cu)
target_link_libraries(cgs_ste_gpu_tests PRIVATE cgs_ste)
target_compile_features(cgs_ste_gpu_tests PRIVATE cxx_std_20 cuda_std_20)

add_library(cgs_reference STATIC reference/reference.cc reference/reference_checks.cc workloads.cc backends/consumer_fixture.cc)
target_compile_features(cgs_reference PUBLIC cxx_std_20)
target_include_directories(cgs_reference PUBLIC
    "${CELLERATOR_ROOT}/include" "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(cgs_reference PUBLIC CUDA::cudart)
add_executable(cgs_reference_smoke reference/reference_smoke.cc)
target_link_libraries(cgs_reference_smoke PRIVATE cgs_reference)
add_library(cgs_teacher STATIC learning/teacher.cu learning/teacher_chain.cu)
target_compile_features(cgs_teacher PUBLIC cxx_std_20 cuda_std_20)
target_include_directories(cgs_teacher PUBLIC
    "${CELLERATOR_ROOT}/include" "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(cgs_teacher PUBLIC CUDA::cudart)
add_library(cgs_consumers STATIC backends/consumers.cu backends/conditions.cu
    "${CELLERATOR_ROOT}/src/gating/cuda_worksets.cu"
    "${CELLERATOR_ROOT}/src/ops/condition/program.cc")
target_compile_features(cgs_consumers PUBLIC cxx_std_20 cuda_std_20)
target_include_directories(cgs_consumers PUBLIC
    "${CELLERATOR_ROOT}/include" "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(cgs_consumers PUBLIC CUDA::cudart)
target_link_libraries(cgs_study PRIVATE cgs_consumers cgs_reference cgs_ste cgs_teacher cgs_linear_library OpenSSL::Crypto)
add_executable(cgs_consumer_tests consumer_tests_main.cc backends/consumer_tests.cu)
target_compile_features(cgs_consumer_tests PRIVATE cxx_std_20 cuda_std_20)
target_link_libraries(cgs_consumer_tests PRIVATE cgs_consumers cgs_reference)
add_executable(cgs_grouping_probe_qualification grouping_probe_qualification_main.cc backends/consumer_tests.cu)
target_compile_features(cgs_grouping_probe_qualification PRIVATE cxx_std_20 cuda_std_20)
target_link_libraries(cgs_grouping_probe_qualification PRIVATE cgs_consumers cgs_reference)
add_executable(cgs_condition_tests condition_tests_main.cc)
target_link_libraries(cgs_condition_tests PRIVATE cgs_consumers cgs_reference)
add_library(cgs_linear_library STATIC backends/grouped_linear_library.cu
    backends/consumer_tests.cu "${CELLERATOR_ROOT}/src/memory/cuda/resources.cu")
target_compile_features(cgs_linear_library PUBLIC cxx_std_20 cuda_std_20)
target_link_libraries(cgs_linear_library PUBLIC cgs_consumers cgs_reference CUDA::cublas)
add_executable(cgs_linear_library_tests library_tests_main.cc)
target_link_libraries(cgs_linear_library_tests PRIVATE cgs_linear_library)
add_executable(cgs_campaign_layout_test campaign_layout_test.cc campaign_runner.cu data/replay.cc)
target_compile_features(cgs_campaign_layout_test PRIVATE cxx_std_20 cuda_std_20)
target_link_libraries(cgs_campaign_layout_test PRIVATE cgs_consumers cgs_reference cgs_ste cgs_teacher cgs_linear_library OpenSSL::Crypto)
add_executable(cgs_replay_timing_test data/replay_timing_test.cc data/replay.cc)
target_compile_features(cgs_replay_timing_test PRIVATE cxx_std_20)
target_link_libraries(cgs_replay_timing_test PRIVATE cgs_reference OpenSSL::Crypto)
set(CGS_REPLAY_PAYLOAD "${CMAKE_CURRENT_LIST_DIR}/data/cache/pbmc-atac-cgatac01/accessibility.cgatac"
    CACHE FILEPATH "Optional pinned accessibility replay for host-only tests")
if(OpenSSL_FOUND)
  add_executable(cgs_verify_atac data/verify_atac.cc)
  target_compile_features(cgs_verify_atac PRIVATE cxx_std_20)
  target_link_libraries(cgs_verify_atac PRIVATE OpenSSL::Crypto)
endif()
if(HDF5_FOUND AND OpenSSL_FOUND)
  add_executable(cgs_prepare_atac data/prepare_atac.cu)
  target_compile_features(cgs_prepare_atac PRIVATE cxx_std_20 cuda_std_20)
  target_include_directories(cgs_prepare_atac PRIVATE
      "${CELLERATOR_ROOT}/include" "${CELLERATOR_ROOT}/components/CellShard/include" ${HDF5_INCLUDE_DIRS})
  target_link_libraries(cgs_prepare_atac PRIVATE ${HDF5_LIBRARIES} OpenSSL::Crypto CUDA::cudart)
else()
  message(STATUS "cgs_prepare_atac unavailable: HDF5 C and OpenSSL Crypto are required")
endif()
add_executable(cgs_teacher_host_tests learning/teacher_host_test.cc)
target_link_libraries(cgs_teacher_host_tests PRIVATE cgs_teacher)
add_executable(cgs_learning_tests learning_tests_main.cc
    learning/ste_tests.cu learning/teacher_test.cu learning/teacher_chain_test.cu)
target_compile_features(cgs_learning_tests PRIVATE cxx_std_20 cuda_std_20)
target_link_libraries(cgs_learning_tests PRIVATE cgs_ste cgs_teacher cgs_reference)
if(BUILD_TESTING)
  add_test(NAME cgs_contract_smoke COMMAND cgs_contract_smoke)
  add_test(NAME cgs_ste_host_tests COMMAND cgs_ste_host_tests)
  add_test(NAME cgs_teacher_host_tests COMMAND cgs_teacher_host_tests)
  add_test(NAME cgs_reference_smoke COMMAND cgs_reference_smoke)
  add_test(NAME cgs_campaign_layout_test COMMAND cgs_campaign_layout_test)
  if(EXISTS "${CGS_REPLAY_PAYLOAD}")
    add_test(NAME cgs_replay_timing_test COMMAND cgs_replay_timing_test "${CGS_REPLAY_PAYLOAD}")
  else()
    add_test(NAME cgs_replay_timing_test COMMAND "${CMAKE_COMMAND}" -E echo "SKIP: pinned replay payload unavailable")
    set_tests_properties(cgs_replay_timing_test PROPERTIES SKIP_REGULAR_EXPRESSION "SKIP: pinned replay payload unavailable")
  endif()
endif()
