get_filename_component(_nf1_numeric_root "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
if(NOT TARGET cellerator_relation_semantics)
  add_library(cellerator_relation_semantics STATIC "${_nf1_numeric_root}/src/compute/operation/relation_semantics.cc")
  target_include_directories(cellerator_relation_semantics PUBLIC "${_nf1_numeric_root}/include")
  target_compile_features(cellerator_relation_semantics PUBLIC cxx_std_17)
  add_library(Cellerator::relation_semantics ALIAS cellerator_relation_semantics)
endif()
add_library(cellerator_native_numeric STATIC "${CMAKE_CURRENT_LIST_DIR}/host_relation.cc")
if(CMAKE_CUDA_COMPILER)
  target_sources(cellerator_native_numeric PRIVATE "${CMAKE_CURRENT_LIST_DIR}/device_linear.cu")
  target_link_libraries(cellerator_native_numeric PUBLIC CUDA::cudart)
  set_target_properties(cellerator_native_numeric PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
endif()
add_library(Cellerator::native_numeric ALIAS cellerator_native_numeric)
target_include_directories(cellerator_native_numeric PUBLIC "${_nf1_numeric_root}/include")
target_compile_features(cellerator_native_numeric PUBLIC cxx_std_20)
target_link_libraries(cellerator_native_numeric PUBLIC Cellerator::relation_semantics)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(cellerator_native_numeric PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-Werror;-fno-fast-math;-ffp-contract=off>)
endif()
if(CMAKE_CUDA_COMPILER AND (CELLERATOR_BUILD_TESTS OR CELLERATOR_BUILD_SEMANTIC_SPINE_V1))
  add_executable(ceNativeArithmeticTest "${_nf1_numeric_root}/tests/native_numeric/device_arithmetic.cc")
  target_link_libraries(ceNativeArithmeticTest PRIVATE Cellerator::native_numeric)
  target_compile_features(ceNativeArithmeticTest PRIVATE cxx_std_20)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(ceNativeArithmeticTest PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
  endif()
  if(CELLERATOR_BUILD_TESTS)
    add_test(NAME ceNativeArithmeticTest COMMAND ceNativeArithmeticTest)
    set_tests_properties(ceNativeArithmeticTest PROPERTIES LABELS "native;cuda;numeric" TIMEOUT 90)
  endif()
endif()
