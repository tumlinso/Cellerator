get_filename_component(_ce_pack_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_library(cellerator_packing_strategy STATIC strategy.cc)
add_library(Cellerator::packing_strategy ALIAS cellerator_packing_strategy)
target_include_directories(cellerator_packing_strategy PUBLIC "${_ce_pack_root}/include")
target_compile_features(cellerator_packing_strategy PUBLIC cxx_std_20)
# Host SDK uses actual incidence only; full native builds retain their original owner.
if(TARGET Cellerator::indexed_incidence)
  set(_ce_pack_incidence Cellerator::indexed_incidence)
elseif(TARGET Cellerator::indexed_mechanism)
  set(_ce_pack_incidence Cellerator::indexed_mechanism)
else()
  message(FATAL_ERROR "packing_strategy requires the existing indexed incidence owner")
endif()
target_link_libraries(cellerator_packing_strategy PUBLIC Cellerator::native_numeric ${_ce_pack_incidence} cellpack)
unset(_ce_pack_incidence)
unset(_ce_pack_root)
