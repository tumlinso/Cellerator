cmake_minimum_required(VERSION 3.22)
project(cellerator_moonshot_seeds LANGUAGES CXX)
option(MOONSHOT_CUDA "Build unqualified sm70 prototype kernels" OFF)
add_executable(moonshot_layout tests/layout.cc)
target_include_directories(moonshot_layout PRIVATE include)
target_compile_features(moonshot_layout PRIVATE cxx_std_17)
if(MOONSHOT_CUDA)
  enable_language(CUDA)
  if(CMAKE_CUDA_COMPILER_VERSION VERSION_GREATER_EQUAL 13.0)
    message(FATAL_ERROR "Volta offline compilation needs a CUDA 12.x toolchain, not CUDA 13+")
  endif()
  add_library(moonshot_cuda STATIC kernels/quad_mma_sm70.cu kernels/patch16_sm70.cu kernels/product2_jvp_sm70.cu)
  set_target_properties(moonshot_cuda PROPERTIES CUDA_ARCHITECTURES 70 CUDA_STANDARD 17)
  target_include_directories(moonshot_cuda PUBLIC include)
  # No --use_fast_math. Numeric-policy changes must be explicit experiments.
  add_executable(moonshot_gpu_smoke tests/gpu_smoke.cu)
  set_target_properties(moonshot_gpu_smoke PROPERTIES CUDA_ARCHITECTURES 70 CUDA_STANDARD 17)
  target_link_libraries(moonshot_gpu_smoke PRIVATE moonshot_cuda)
endif()
