# Internal consumer of the linked relation core and bounded compiler adapter.
# These build targets do not define an installed SDK.
cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(RU1Reference LANGUAGES CXX)
  option(CELLERATOR_RU1_REFERENCE_ONLY "Build independent fixture only" ON)
endif()
if(CELLERATOR_RU1_REFERENCE_ONLY)
  add_executable(ceRelationUpdateReference regulatory_learning.cc)
  target_compile_definitions(ceRelationUpdateReference PRIVATE CELLERATOR_RU1_REFERENCE_ONLY=1)
  target_compile_features(ceRelationUpdateReference PRIVATE cxx_std_20)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(ceRelationUpdateReference PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
  endif()
  enable_testing()
  add_test(NAME ru1_reference_fixture COMMAND ceRelationUpdateReference)
else()
  if(NOT TARGET cellerator_relation_update_core OR NOT TARGET cellerator_relation_update_compiler)
    message(FATAL_ERROR "RU1 requires the linked native core and compiler targets")
  endif()
  find_package(CUDAToolkit REQUIRED)
  add_executable(ceRelationUpdateDemo regulatory_learning.cc)
  target_compile_features(ceRelationUpdateDemo PRIVATE cxx_std_20)
  target_link_libraries(ceRelationUpdateDemo PRIVATE cellerator_relation_update_core cellerator_relation_update_compiler CUDA::cudart)
  add_test(NAME ru1_demo_hybrid COMMAND ceRelationUpdateDemo --route hybrid)
  add_test(NAME ru1_demo_sparse COMMAND ceRelationUpdateDemo --route sparse)
endif()
