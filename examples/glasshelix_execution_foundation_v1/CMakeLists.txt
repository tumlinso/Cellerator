cmake_minimum_required(VERSION 3.20)
project(CelleratorNF1Consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT TARGET Cellerator::native_foundation)
 find_package(Cellerator CONFIG REQUIRED COMPONENTS native_foundation)
endif()
add_executable(combinatorial_consumer combinatorial_consumer.cc)
target_link_libraries(combinatorial_consumer PRIVATE Cellerator::native_foundation)
enable_testing()
add_test(NAME ce_nf1_standalone_consumer_cuda COMMAND combinatorial_consumer)
