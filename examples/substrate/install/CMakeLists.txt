cmake_minimum_required(VERSION 3.24)
project(CelleratorInstalledConsumer LANGUAGES CXX)
include(CTest)
option(CE_REQUIRE_INTEGRATED "Run the native integrated vertical consumer" OFF)
option(CE_EFFECTS_ONLY "Request only the native host effects component" OFF)
option(CE_SEQUENCE_ONLY "Request only the native sequence component" OFF)
option(CE_SEQUENCE_MODULE "Import the installed sequence module" OFF)
option(CE_SEQUENCE_CUDA "Link a CUDA sequence launcher from a C++ consumer" OFF)
if(CE_EFFECTS_ONLY)
  find_package(Cellerator CONFIG REQUIRED COMPONENTS effects)
  if(TARGET Cellerator::native_numeric OR TARGET CUDA::cudart OR TARGET Torch::Torch)
    message(FATAL_ERROR "Effects-only import pulled unrelated numerical/adapter dependencies")
  endif()
  add_executable(ce_installed_consumer effects_only.cc)
  target_link_libraries(ce_installed_consumer PRIVATE Cellerator::effects)
elseif(CE_SEQUENCE_ONLY)
  find_package(Cellerator CONFIG REQUIRED COMPONENTS sequence)
  if(TARGET Cellerator::native_numeric OR TARGET Baseplane::seq OR TARGET Torch::Torch)
    message(FATAL_ERROR "Sequence-only import pulled unrelated dependencies")
  endif()
  if(CE_SEQUENCE_CUDA AND NOT TARGET CUDA::cudart)
    message(FATAL_ERROR "CUDA sequence component did not resolve its runtime dependency")
  elseif(NOT CE_SEQUENCE_CUDA AND TARGET CUDA::cudart)
    message(FATAL_ERROR "Host sequence component pulled CUDA runtime")
  endif()
  add_executable(ce_installed_consumer sequence_only.cc)
  target_link_libraries(ce_installed_consumer PRIVATE Cellerator::sequence)
  if(CE_SEQUENCE_MODULE)
    target_compile_definitions(ce_installed_consumer PRIVATE CE_SEQUENCE_MODULE)
    set_target_properties(ce_installed_consumer PROPERTIES CXX_EXTENSIONS OFF CXX_SCAN_FOR_MODULES ON)
  endif()
  if(CE_SEQUENCE_CUDA)
    target_compile_definitions(ce_installed_consumer PRIVATE CE_SEQUENCE_CUDA)
  endif()
else()
  set(components host_operation_contracts native_numeric executable_program indexed_incidence cellpack)
  if(CE_REQUIRE_INTEGRATED)
    list(APPEND components structured_state packing_strategy matrix_patch private_ports effects prepared_host)
  endif()
  find_package(Cellerator CONFIG REQUIRED COMPONENTS ${components})
  find_package(Cellerator CONFIG REQUIRED COMPONENTS relation_semantics) # Already imported dependency.
  add_executable(ce_installed_consumer consumer.cc)
  foreach(component IN LISTS components)
    target_link_libraries(ce_installed_consumer PRIVATE Cellerator::${component})
    get_target_property(capabilities Cellerator::${component} CELLERATOR_CAPABILITIES)
    if(NOT capabilities)
      message(FATAL_ERROR "Component ${component} lacks explicit capability metadata")
    endif()
    get_target_property(links Cellerator::${component} INTERFACE_LINK_LIBRARIES)
    if(links MATCHES "Baseplane::|moonshot_")
      message(FATAL_ERROR "CPU Cellerator owner still links a retired component")
    endif()
  endforeach()
  if(CE_REQUIRE_INTEGRATED)
    target_sources(ce_installed_consumer PRIVATE integrated.cc)
    target_compile_definitions(ce_installed_consumer PRIVATE CE_REQUIRE_INTEGRATED=1)
  endif()
endif()
target_compile_features(ce_installed_consumer PRIVATE cxx_std_20)
target_compile_options(ce_installed_consumer PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
add_test(NAME ce_installed_consumer COMMAND ce_installed_consumer)
