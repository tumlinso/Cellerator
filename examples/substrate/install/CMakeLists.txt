cmake_minimum_required(VERSION 3.24)
project(CelleratorInstalledConsumer LANGUAGES CXX)
include(CTest)
option(CE_REQUIRE_INTEGRATED "Run the native integrated vertical consumer" OFF)
option(CE_EFFECTS_ONLY "Request only the independent experimental host effects component" OFF)
if(CE_EFFECTS_ONLY)
  find_package(Cellerator CONFIG REQUIRED COMPONENTS moonshot_effects)
  if(TARGET Cellerator::native_numeric OR TARGET CUDA::cudart OR TARGET Torch::Torch)
    message(FATAL_ERROR "Effects-only import pulled unrelated numerical/adapter dependencies")
  endif()
  add_executable(ce_installed_consumer effects_only.cc)
  target_link_libraries(ce_installed_consumer PRIVATE Cellerator::moonshot_effects)
else()
  set(components host_operation_contracts native_numeric executable_program indexed_incidence cellpack
    moonshot_effects moonshot_mechanisms moonshot_learning)
  if(CE_REQUIRE_INTEGRATED)
    list(APPEND components structured_state packing_strategy matrix_patch private_ports effects prepared_host)
  endif()
  find_package(Cellerator CONFIG REQUIRED COMPONENTS ${components})
  find_package(Cellerator CONFIG REQUIRED COMPONENTS relation_semantics) # Already imported dependency.
  add_executable(ce_installed_consumer consumer.cc)
  foreach(component IN LISTS components)
    target_link_libraries(ce_installed_consumer PRIVATE Cellerator::${component})
    get_target_property(capabilities Cellerator::${component} CELLERATOR_CAPABILITIES)
    if(NOT capabilities)
      message(FATAL_ERROR "Component ${component} lacks explicit capability metadata")
    endif()
    get_target_property(links Cellerator::${component} INTERFACE_LINK_LIBRARIES)
    if(links MATCHES "Baseplane::")
      message(FATAL_ERROR "CPU CE owner has an unexpected back-edge into Baseplane")
    endif()
  endforeach()
  if(CE_REQUIRE_INTEGRATED)
    target_sources(ce_installed_consumer PRIVATE integrated.cc)
    target_compile_definitions(ce_installed_consumer PRIVATE CE_REQUIRE_INTEGRATED=1)
  endif()
endif()
target_compile_features(ce_installed_consumer PRIVATE cxx_std_20)
target_compile_options(ce_installed_consumer PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
add_test(NAME ce_installed_consumer COMMAND ce_installed_consumer)
