cmake_minimum_required(VERSION 3.28)
project(velocity_cuda_consumer LANGUAGES CXX CUDA)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CUDA_STANDARD 20)
find_package(anndata CONFIG REQUIRED)
if(TARGET CUDA::cudart)
  message(FATAL_ERROR "CPU-only package discovery unexpectedly loaded CUDA")
endif()
find_package(velocity CONFIG REQUIRED COMPONENTS cuda)
add_library(device_abi OBJECT device_abi.cu)
target_link_libraries(device_abi PRIVATE anndata::cuda CUDA::cudart)
foreach(name IN ITEMS cuda_consumer cuda_consumer_again)
  add_executable(${name} main.cpp $<TARGET_OBJECTS:device_abi>)
  target_link_libraries(${name} PRIVATE velocity::cuda CUDA::cudart)
endforeach()
