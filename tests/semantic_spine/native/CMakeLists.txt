cmake_minimum_required(VERSION 3.21)
project(cellerator_semantic_spine_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 70)
get_filename_component(SPINE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
# Build the actual current dependency graph. The old handwritten archive list
# requested a nonexistent abi archive and omitted transitive geometry/runtime.
# No borrowed archive can silently certify a different prepared-pair source.
set(CELLERATOR_ENABLE_CUDA ON CACHE STRING "" FORCE)
set(CELLERATOR_BUILD_SEMANTIC_SPINE_V1 ON CACHE BOOL "" FORCE)
set(CELLERATOR_BUILD_RELATION_UPDATE_SPINE_V1 OFF CACHE BOOL "" FORCE)
set(CELLERATOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SPINE_ROOT}" foundation EXCLUDE_FROM_ALL)
enable_language(CUDA)
find_package(CUDAToolkit 12.9 EXACT REQUIRED)
foreach(name adapter preparation generation forward transpose lifecycle overhead)
    add_executable(spine_${name}_test "${name}_test.cu")
    target_include_directories(spine_${name}_test PRIVATE "${SPINE_ROOT}/include")
    target_compile_options(spine_${name}_test PRIVATE -UNDEBUG)
    target_link_libraries(spine_${name}_test PRIVATE cellerator_relation_update_support)
endforeach()
add_executable(spine_regulatory_reuse "${SPINE_ROOT}/examples/semantic_spine_v1/regulatory_reuse.cc")
target_link_libraries(spine_regulatory_reuse PRIVATE Cellerator::prepared_relation_cuda)
