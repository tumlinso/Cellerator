cmake_minimum_required(VERSION 3.20)
project(CelleratorCellConsumer LANGUAGES CXX)

option(CELLERATOR_CELL_TEST_NN "Also validate installed native NN and relation linkage" OFF)
if(CELLERATOR_CELL_TEST_NN)
    find_package(Cellerator CONFIG REQUIRED COMPONENTS native_data nn)
else()
    find_package(Cellerator CONFIG REQUIRED)
endif()
if(NOT CELLERATOR_CELL_TEST_NN AND
   (TARGET Cellerator::nn OR TARGET torch OR TARGET torch_cpu))
    message(FATAL_ERROR "The borrowed-view consumer must not discover Torch/NN")
endif()

enable_testing()
foreach(example IN ITEMS event_forward multimodal)
    set(target "cell_consumer_${example}")
    add_executable(${target} "../../../examples/native_cell/${example}.cc")
    target_link_libraries(${target} PRIVATE Cellerator::headers)
    add_test(NAME ${target} COMMAND ${target})
endforeach()
if(CELLERATOR_CELL_TEST_NN)
    add_executable(cell_consumer_nn nn.cc)
    target_link_libraries(cell_consumer_nn PRIVATE Cellerator::nn)
    target_compile_features(cell_consumer_nn PRIVATE cxx_std_20)
    add_test(NAME cell_consumer_nn COMMAND cell_consumer_nn)
    add_executable(cell_consumer_relation ../../framework/native/cell_relation_example.cc)
    target_link_libraries(cell_consumer_relation PRIVATE Cellerator::native_data)
    add_test(NAME cell_consumer_relation COMMAND cell_consumer_relation)
endif()
