cmake_minimum_required(VERSION 3.20)
project(CelleratorCellConsumer LANGUAGES CXX)

option(CELLERATOR_CELL_TEST_NN "Also validate installed native NN/modeling linkage" OFF)
if(CELLERATOR_CELL_TEST_NN)
    find_package(Cellerator CONFIG REQUIRED COMPONENTS modeling nn)
else()
    find_package(Cellerator CONFIG REQUIRED)
endif()
if(NOT CELLERATOR_CELL_TEST_NN AND
   (TARGET Cellerator::nn OR TARGET torch OR TARGET torch_cpu))
    message(FATAL_ERROR "The native Cell consumer must not discover Torch/NN")
endif()

enable_testing()
foreach(part core adapters)
    add_executable(cell_consumer_${part} ${part}.cc)
    target_link_libraries(cell_consumer_${part} PRIVATE Cellerator::libCellerator)
    set_target_properties(cell_consumer_${part} PROPERTIES CXX_STANDARD_REQUIRED YES)
    add_test(NAME cell_consumer_${part} COMMAND cell_consumer_${part})
endforeach()
set_target_properties(cell_consumer_core PROPERTIES CXX_STANDARD 20)
set_target_properties(cell_consumer_adapters PROPERTIES CXX_STANDARD 20)
if(CELLERATOR_CELL_TEST_NN)
    add_executable(cell_consumer_nn nn.cc)
    target_link_libraries(cell_consumer_nn PRIVATE Cellerator::nn Cellerator::modeling)
    target_compile_features(cell_consumer_nn PRIVATE cxx_std_20)
    add_test(NAME cell_consumer_nn COMMAND cell_consumer_nn)
endif()
