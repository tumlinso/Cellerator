cmake_minimum_required(VERSION 3.20)
if(NF1_VALUES_OUTER_BUILD)
  return()
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(CelleratorNativeValuesQualification LANGUAGES CXX)
  enable_testing()
  set(NF1_VALUES_OUTER_BUILD ON)
  set(CELLERATOR_BUILD_SEMANTIC_SPINE_V1 ON CACHE BOOL "Required actual pair owner" FORCE)
  set(CELLERATOR_BUILD_TESTS OFF CACHE BOOL "Only scoped qualification" FORCE)
  get_filename_component(_nf1_values_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
  add_subdirectory("${_nf1_values_root}" "${CMAKE_CURRENT_BINARY_DIR}/cellerator")
  enable_language(CUDA)
endif()

  message(FATAL_ERROR "Values qualification requires the real prepared relation CUDA owner")

endif()
add_executable(ce_nf1_v01 v01.cu)

set_target_properties(ce_nf1_v01 PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
add_test(NAME ce_nf1_v01 COMMAND ce_nf1_v01)
set_tests_properties(ce_nf1_v01 PROPERTIES LABELS "nf1;cuda;values;sm70" TIMEOUT 60)

  --tool memcheck --error-exitcode 99 $<TARGET_FILE:ce_nf1_v01>)
