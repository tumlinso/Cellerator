cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(CelleratorLifetimeConformance LANGUAGES CXX)
  enable_testing()
  get_filename_component(_nf1_default_owner "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
  set(NF1_OWNER_ROOT "${_nf1_default_owner}" CACHE PATH "Integrated or explicitly pinned production owner")
  set(CELLERATOR_BUILD_TESTS OFF CACHE BOOL "Scoped conformance only" FORCE)
  set(CELLERATOR_BUILD_SEMANTIC_SPINE_V1 ON CACHE BOOL "Actual retained relation owner" FORCE)
  add_subdirectory("${NF1_OWNER_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/owner")
  enable_language(CUDA)
endif()
if(NOT TARGET Cellerator::prepared_relation_cuda)
  message(FATAL_ERROR "T03 requires the source-linked production relation/readiness owner")
endif()
add_executable(ce_nf1_t03 t03.cu)
target_link_libraries(ce_nf1_t03 PRIVATE Cellerator::prepared_relation_cuda)
set_target_properties(ce_nf1_t03 PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
add_test(NAME ce_nf1_t03 COMMAND ce_nf1_t03)
set_tests_properties(ce_nf1_t03 PROPERTIES LABELS "nf1;cuda;lifetime;independent" TIMEOUT 90)
find_program(NF1_COMPUTE_SANITIZER compute-sanitizer HINTS "${CUDAToolkit_ROOT}/compute-sanitizer" REQUIRED)
add_test(NAME ce_nf1_t03_memcheck COMMAND "${NF1_COMPUTE_SANITIZER}" --tool memcheck --error-exitcode 99 $<TARGET_FILE:ce_nf1_t03>)
set_tests_properties(ce_nf1_t03_memcheck PROPERTIES LABELS "nf1;cuda;sanitizer" TIMEOUT 180)
