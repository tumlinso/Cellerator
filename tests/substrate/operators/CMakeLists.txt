cmake_minimum_required(VERSION 3.18)
project(CelleratorNativeOperators LANGUAGES CXX)
include(CTest)
get_filename_component(CE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_subdirectory("${CE_ROOT}/src/math/matrix" matrix-owner)
add_subdirectory("${CE_ROOT}/src/math/process" process-owner)
add_executable(ce_is1_native_operators native_consumer.cc)
target_link_libraries(ce_is1_native_operators PRIVATE Cellerator::matrix_patch Cellerator::private_ports)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  foreach(t cellerator_matrix_patch cellerator_private_ports ce_is1_native_operators)
    target_compile_options(${t} PRIVATE -Wall -Wextra -Werror -pedantic -fno-fast-math -ffp-contract=off)
  endforeach()
endif()
add_test(NAME ce_is1_native_operators COMMAND ce_is1_native_operators)
