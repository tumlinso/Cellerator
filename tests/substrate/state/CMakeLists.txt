cmake_minimum_required(VERSION 3.18)
project(CelleratorStructuredState LANGUAGES CXX)
include(CTest)
get_filename_component(CE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
if(NOT TARGET Cellerator::structured_state)
  add_subdirectory("${CE_ROOT}/src/state" "state-owner")
endif()
add_executable(ce_is1_structured_state structured_state_test.cc)
target_link_libraries(ce_is1_structured_state PRIVATE Cellerator::structured_state)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(cellerator_structured_state PRIVATE -Wall -Wextra -Werror -pedantic)
  target_compile_options(ce_is1_structured_state PRIVATE -Wall -Wextra -Werror -pedantic -fno-fast-math -ffp-contract=off)
endif()
add_test(NAME ce_is1_structured_state COMMAND ce_is1_structured_state)
