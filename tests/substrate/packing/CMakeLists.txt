cmake_minimum_required(VERSION 3.18)
project(CelleratorPackingContract LANGUAGES CXX)
include(CTest)
get_filename_component(CE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
# Actual existing implementations are source-qualified for this host-only fixture.
# Native integration links their full component targets when configured.
add_subdirectory("${CE_ROOT}/src/compute/operation/native_numeric" native-numeric)
if(NOT TARGET Cellerator::indexed_incidence AND NOT TARGET Cellerator::indexed_mechanism)
  add_library(ce_pack_host_incidence STATIC "${CE_ROOT}/src/compute/operation/indexed_mechanism/incidence.cc")
  target_include_directories(ce_pack_host_incidence PUBLIC "${CE_ROOT}/include")
  target_compile_features(ce_pack_host_incidence PUBLIC cxx_std_20)
  add_library(Cellerator::indexed_incidence ALIAS ce_pack_host_incidence)
endif()
if(NOT TARGET cellpack)
  add_library(cellpack STATIC "${CE_ROOT}/src/geometry/evaluator.cc" "${CE_ROOT}/src/geometry/planner.cc")
  target_include_directories(cellpack PUBLIC "${CE_ROOT}/include")
  target_compile_features(cellpack PUBLIC cxx_std_17)
endif()
add_subdirectory("${CE_ROOT}/src/packing/core" packing-owner)
add_executable(ce_is1_packing_contract strategy_test.cc)
target_link_libraries(ce_is1_packing_contract PRIVATE Cellerator::packing_strategy)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  foreach(target cellerator_packing_strategy ce_is1_packing_contract cellpack)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic -fno-fast-math -ffp-contract=off)
  endforeach()
endif()
add_test(NAME ce_is1_packing_contract COMMAND ce_is1_packing_contract)
