cmake_minimum_required(VERSION 3.24)
project(cellerator_moonshot_mma LANGUAGES CXX)
option(MOONSHOT_CUDA "Compile SM70 prototypes and admission/device smoke binaries" OFF)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME moonshot_mma_cpu COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/check_mma.py)
if(MOONSHOT_CUDA)
  enable_language(CUDA)
  if(CMAKE_CUDA_COMPILER_VERSION VERSION_GREATER_EQUAL 13.0 OR CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 12.9)
    message(FATAL_ERROR "These SM70 prototypes require the reviewed CUDA 12.9 toolchain")
  endif()
  find_package(CUDAToolkit 12.9 REQUIRED)
  set(CMAKE_CUDA_ARCHITECTURES 70)
  add_subdirectory(patch16)
  add_subdirectory(quad)
endif()
add_subdirectory(product)
