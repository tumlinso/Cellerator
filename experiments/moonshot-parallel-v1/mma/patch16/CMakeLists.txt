cmake_minimum_required(VERSION 3.24)
project(moonshot_patch16 LANGUAGES CXX CUDA)
add_library(moonshot_patch16 STATIC patch16.cu)
target_include_directories(moonshot_patch16 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(moonshot_patch16 PUBLIC cxx_std_17 cuda_std_17)
set_target_properties(moonshot_patch16 PROPERTIES CUDA_ARCHITECTURES 70)
add_executable(patch16_smoke smoke.cu)
target_link_libraries(patch16_smoke PRIVATE moonshot_patch16)
set_target_properties(patch16_smoke PROPERTIES CUDA_ARCHITECTURES 70)
enable_testing()
add_test(NAME patch16_admission COMMAND patch16_smoke --host-only)
add_test(NAME patch16_gpu COMMAND patch16_smoke)
set_tests_properties(patch16_gpu PROPERTIES SKIP_RETURN_CODE 77)
