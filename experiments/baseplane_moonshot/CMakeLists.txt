cmake_minimum_required(VERSION 3.18)
project(cellerator_moonshot LANGUAGES CXX)

add_library(ce_moonshot INTERFACE)
add_library(Cellerator::moonshot ALIAS ce_moonshot)
target_include_directories(ce_moonshot INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(ce_moonshot INTERFACE cxx_std_17)

option(CE_MOON_ENABLE_CUDA "Compile experimental kernels; does not launch a GPU" OFF)
if(CE_MOON_ENABLE_CUDA)
  enable_language(CUDA)
  add_library(ce_moon_cuda_compile OBJECT cuda/compile.cu)
  target_link_libraries(ce_moon_cuda_compile PRIVATE Cellerator::moonshot)
  set_target_properties(ce_moon_cuda_compile PROPERTIES CUDA_ARCHITECTURES 70 CUDA_STANDARD 17)
  add_executable(ce_moon_cuda_header_link cuda/link_a.cu cuda/link_b.cu)
  target_link_libraries(ce_moon_cuda_header_link PRIVATE Cellerator::moonshot)
  set_target_properties(ce_moon_cuda_header_link PROPERTIES CUDA_ARCHITECTURES 70 CUDA_STANDARD 17)
endif()

enable_testing()
add_executable(ce_moon_host_smoke host/smoke.cpp)
target_link_libraries(ce_moon_host_smoke PRIVATE Cellerator::moonshot)
add_test(NAME ce_moon_host_smoke COMMAND ce_moon_host_smoke)
find_package(Python3 QUIET COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME ce_moon_learn_lut COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/python/learn_lut.py)
endif()

# Family owners add their own targets/tests without editing this shared file.
file(GLOB ce_moon_family_cmake CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/families/*/CMakeLists.txt")
foreach(family_cmake IN LISTS ce_moon_family_cmake)
  get_filename_component(family_dir "${family_cmake}" DIRECTORY)
  get_filename_component(family_name "${family_dir}" NAME)
  add_subdirectory("${family_dir}" "families/${family_name}")
endforeach()
